create or replace function public.cap_nhat_ton_va_gia_von()
returns trigger
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_gia_von_cu numeric(18,4);
  v_gia_von_moi numeric(18,4);
  v_ton_cu numeric(18,4);
begin
  -- BƯỚC 1 — KHÓA DÒNG san_pham TRƯỚC MỌI THỨ KHÁC.
  -- Đây là điểm serialize duy nhất. Đảo thứ tự bước 1 và bước 2 sẽ tạo lost update:
  -- hai phiếu nhập cùng mã chạy gần như đồng thời, cả hai đọc tồn/giá cũ trước khi
  -- cái kia commit, một bản ghi cost bị mất. Lỗi này KHÔNG lộ ra trong test
  -- một-transaction, chỉ lộ dưới tải thật.
  select gia_von into v_gia_von_cu
  from public.san_pham
  where id = new.san_pham_id
  for update;

  -- BƯỚC 2 — đọc tồn cũ toàn công ty (an toàn vì đã giữ khóa ở bước 1).
  select coalesce(sum(so_luong), 0) into v_ton_cu
  from public.ton_kho
  where san_pham_id = new.san_pham_id;

  -- BƯỚC 3 — cập nhật tồn của đúng kho phát sinh.
  insert into public.ton_kho (kho_id, san_pham_id, so_luong, cap_nhat_luc)
  values (new.kho_id, new.san_pham_id, new.so_luong, now())
  on conflict (kho_id, san_pham_id)
  do update set
    so_luong = public.ton_kho.so_luong + excluded.so_luong,
    cap_nhat_luc = now();

  -- BƯỚC 4 — giá vốn chỉ tính lại khi NHẬP (so_luong > 0). Xuất không đổi giá vốn.
  if new.so_luong > 0 then
    if (v_ton_cu + new.so_luong) = 0 then
      v_gia_von_moi := v_gia_von_cu;
    else
      v_gia_von_moi := ((v_ton_cu * v_gia_von_cu) + (new.so_luong * new.gia_von_tai_thoi_diem))
                       / (v_ton_cu + new.so_luong);
    end if;
  else
    v_gia_von_moi := v_gia_von_cu;
  end if;

  update public.san_pham
  set gia_von = v_gia_von_moi,
      lan_phat_sinh_cuoi = now()
  where id = new.san_pham_id;

  return new;
end;
$$;

create trigger cap_nhat_ton_sau_khi_ghi_so_cai
  after insert on public.kho_movement
  for each row execute function public.cap_nhat_ton_va_gia_von();
```

Điểm bắt buộc, không được sắp xếp lại:

1. **`for update` trên `san_pham` phải là câu lệnh ĐẦU TIÊN.** Không phải sau khi đọc
   `ton_kho`, không phải cuối hàm. RESEARCH.md Pitfall 4 giải thích đầy đủ.
2. **`security definer` + `set search_path = ''`** — trigger phải ghi được vào `san_pham.gia_von`
   mà client không có quyền UPDATE cột đó (plan 08 sẽ REVOKE). `search_path = ''` chặn lỗ hổng
   leo quyền, nên MỌI tên object trong thân hàm phải schema-qualify (`public.san_pham`).
3. **Không dùng `SERIALIZABLE`** — READ COMMITTED (mặc định) + khóa dòng tường minh là đủ,
   và tránh chi phí retry-on-conflict khó xử lý đúng bên trong trigger.
4. **Xuất (so_luong < 0) không đổi giá vốn** — bình quân gia quyền chỉ tính lại khi nhập.
5. **Chia 0**: khi `v_ton_cu + new.so_luong = 0` thì giữ nguyên giá cũ, không để hàm văng.

Cập nhật `lan_phat_sinh_cuoi` ở đây chính là nguồn dữ liệu cho việc xếp mã phát sinh gần
đây lên trước ở RPC tìm kiếm (plan 07).

-- LỚP 1 — thu hồi quyền.
-- Supabase mặc định chạy ALTER DEFAULT PRIVILEGES GRANT ALL ON TABLES TO
-- postgres, anon, authenticated, service_role, nên mọi bảng mới TỰ ĐỘNG có
-- UPDATE/DELETE cho cả service_role. REVOKE này KHÔNG thừa.
-- Nhắm đích danh từng role, không REVOKE FROM PUBLIC (quyền được cấp trực tiếp
-- cho từng role, không đi qua PUBLIC).
revoke update, delete on public.kho_movement from anon, authenticated, service_role;

-- LỚP 2 — trigger. Cần lớp này vì REVOKE KHÔNG chặn được chủ sở hữu bảng
-- (role postgres, dùng để chạy migration và trong SQL Editor) — quyền sở hữu
-- không bị REVOKE loại bỏ. Trigger chặn được MỌI role kể cả owner/superuser.
create or replace function public.chan_sua_xoa_so_cai()
returns trigger
language plpgsql
as $$
begin
  raise exception 'Sổ cái kho_movement là bất biến, không được sửa hoặc xóa (thao tác: %)', tg_op
    using errcode = '23514';
end;
$$;

create trigger chan_sua_xoa_kho_movement
  before update or delete on public.kho_movement
  for each row execute function public.chan_sua_xoa_so_cai();
```

`errcode = '23514'` (check_violation) là cố ý, không phải mã tùy tiện: `src/shared/lib/errors.ts`
đã map sẵn `23514` thành thông báo tiếng Việt "Dữ liệu không hợp lệ". Dùng `RAISE EXCEPTION`
mặc định sẽ ra `P0001` và rơi vào nhánh "không xác định", người dùng đọc message thô.

Muốn sửa một movement đã ghi thì cách duy nhất là hủy chứng từ sinh ra nó để tạo bút toán
đảo (plan 06) — đó chính là ý đồ thiết kế, không phải hạn chế.
