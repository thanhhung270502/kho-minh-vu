```sql
create or replace function public.custom_access_token_hook(event jsonb)
returns jsonb
language plpgsql
stable
as $$
declare
  claims jsonb;
  v_vai_tro public.vai_tro;
  v_kho_id uuid;
begin
  select vai_tro, kho_id into v_vai_tro, v_kho_id
  from public.nguoi_dung
  where id = (event->>'user_id')::uuid;

  claims := event->'claims';

  if v_vai_tro is not null then
    claims := jsonb_set(claims, '{vai_tro}', to_jsonb(v_vai_tro));
  end if;
  if v_kho_id is not null then
    claims := jsonb_set(claims, '{kho_id}', to_jsonb(v_kho_id));
  end if;

  return jsonb_build_object('claims', claims);
end;
$$;

grant usage on schema public to supabase_auth_admin;
grant execute on function public.custom_access_token_hook(jsonb) to supabase_auth_admin;
grant select on public.nguoi_dung to supabase_auth_admin;

-- Hook chỉ được GoTrue gọi. Client tuyệt đối không được tự gọi để tự cấp claim.
revoke execute on function public.custom_access_token_hook(jsonb) from authenticated, anon, public;
```

chưa tồn tại thì bỏ comment bây giờ):

token đang sống đổi theo — người dùng vẫn thao tác theo vai trò cũ tới lần refresh kế tiếp.
Phase 2 (CDAT-01, màn Cài đặt) phải gọi `supabase.auth.admin.signOut(userId, 'others')` sau

**AUTH-05 — văn phòng không sửa giá, dùng HAI lớp:**

RLS chặn theo DÒNG, không theo CỘT — không thể dùng policy để chặn riêng `gia_von`.

```sql
-- Lớp 1: quyền theo cột. PostgREST tôn trọng và trả 42501 ngay, không tốn round-trip.
revoke update on public.san_pham from authenticated;
grant update (
  ma_hang, ten_hang, barcode, nhom_hang_id, dvt_id, cong_doan_id, quy_doi,
  ton_toi_thieu, ton_toi_da, hinh_anh_url, vi_tri_ke, dang_kinh_doanh, ghi_chu,
  gia_ban
) on public.san_pham to authenticated;
-- CHỈ gia_von vắng mặt. Nó do trigger cap_nhat_ton_va_gia_von (SECURITY DEFINER)
-- ghi, không client nào được cấp quyền UPDATE cột này.
--
-- gia_ban CÓ trong danh sách, cố ý: quyền theo cột của Postgres áp theo SQL ROLE,
-- không theo JWT claim. Cả quản lý lẫn văn phòng đều kết nối dưới cùng role
-- `authenticated` — chỉ khác nhau ở claim. Bỏ gia_ban khỏi GRANT sẽ chặn luôn
-- quản lý ở tầng privilege, trước khi trigger bên dưới kịp chạy, khiến nhánh
-- "quản lý được sửa" thành code chết. Phân biệt vai trò là việc của trigger.

-- Lớp 2: trigger là lưới an toàn cho đường ghi qua RPC nội bộ.
create or replace function public.chan_sua_gia_khong_du_quyen()
returns trigger
language plpgsql
as $$
begin
  if new.gia_ban is distinct from old.gia_ban
     and coalesce((select public.vai_tro_hien_tai())::text, '') <> 'quan_ly' then
    raise exception 'Chỉ vai trò quản lý được sửa giá bán' using errcode = '42501';
  end if;
  return new;
end;
$$;

create trigger chan_sua_gia_san_pham
  before update on public.san_pham
  for each row execute function public.chan_sua_gia_khong_du_quyen();
```

Trigger KHÔNG kiểm `gia_von` vì chính trigger giá vốn (chạy `security definer`, ngoài ngữ
cảnh JWT) là thứ ghi cột đó — kiểm ở đây sẽ chặn nhầm đường ghi hợp lệ. `gia_von` đã được
bảo vệ hoàn toàn bằng lớp 1 (không nằm trong GRANT cột).

**Phân vai rõ giữa hai lớp:** lớp 1 (GRANT cột) chặn `gia_von` với MỌI client, vĩnh viễn.
Lớp 2 (trigger) là thứ duy nhất phân biệt được vai trò, và nó chỉ gác `gia_ban`.

**Quy tắc bắt buộc cho MỌI policy trong file:** luôn `(select public.vai_tro_hien_tai())`,
không bao giờ gọi trần `public.vai_tro_hien_tai()`. Bọc subquery làm Postgres cache một lần
mỗi câu lệnh thay vì gọi lại mỗi dòng.
