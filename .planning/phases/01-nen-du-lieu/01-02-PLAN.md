```sql
  id uuid primary key default uuid_generate_v4(),
  ma text not null unique,
  ten text not null,
  dia_chi text,

  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- PK CHÍNH LÀ auth.users.id — không sinh uuid mới. Một tài khoản Supabase Auth
-- tương ứng đúng một dòng nguoi_dung.
create table public.nguoi_dung (
  id uuid primary key references auth.users(id) on delete cascade,
  ho_ten text not null,
  vai_tro public.vai_tro not null default 'chi_xem',
  kho_id uuid references public.kho(id),
  dang_hoat_dong boolean not null default true,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

create index idx_nguoi_dung_kho on public.nguoi_dung (kho_id) where kho_id is not null;

create trigger set_updated_at_kho
  before update on public.kho
  for each row execute function public.update_updated_at();

create trigger set_updated_at_nguoi_dung
  before update on public.nguoi_dung
  for each row execute function public.update_updated_at();
```

`kho_id` nullable có chủ đích: vai trò `quan_ly` và `van_phong` không gắn kho cố định,
chỉ `thu_kho` mới bắt buộc có (ràng buộc đó kiểm ở tầng ứng dụng/Cài đặt ở Phase 2, không

`on delete cascade` trên FK tới auth.users là ngoại lệ duy nhất của dự án: xóa tài khoản

Thêm vào cuối `supabase/migrations/0003_nguoi_dung_kho.sql`:

```sql
-- Đọc vai trò từ JWT claim, KHÔNG truy vấn bảng nguoi_dung.
-- Policy phải `select vai_tro from nguoi_dung where id = auth.uid()` sẽ chạy
-- MỖI DÒNG — quét bảng tồn 3.266 mã sẽ chậm thấy rõ. Đây là lý do D-15 tồn tại.
create or replace function public.vai_tro_hien_tai()
returns public.vai_tro
language sql
stable
security definer
set search_path = ''
as $$
  select nullif(auth.jwt() ->> 'vai_tro', '')::public.vai_tro;
$$;

create or replace function public.kho_hien_tai()
returns uuid
language sql
stable
security definer
set search_path = ''
as $$
  select nullif(auth.jwt() ->> 'kho_id', '')::uuid;
$$;

grant execute on function public.vai_tro_hien_tai() to authenticated;
grant execute on function public.kho_hien_tai() to authenticated;
```

**Vì sao hai hàm này nằm ở 0003 chứ không ở 0014 (migration RLS):** RPC `ghi_so_chung_tu`
và `huy_chung_tu` ở migration 0011/0012 gọi `vai_tro_hien_tai()` để kiểm quyền nghiệp vụ
(SECURITY DEFINER bỏ qua RLS nên phải kiểm tường minh). Đặt helper ở 0014 tạo phụ thuộc
ngược — migration 0011 chạy trước 0014. Helper chỉ cần enum `vai_tro` từ 0002 nên đặt ở
0003 là sớm nhất hợp lệ.

**Cách dùng bắt buộc trong policy:** LUÔN bọc `(select public.vai_tro_hien_tai())`, không
bao giờ gọi trần. Bọc trong subquery làm Postgres cache kết quả một lần mỗi câu lệnh
(initPlan) thay vì gọi lại mỗi dòng — dù bản thân hàm đã `stable`.
