```sql
create table public.chuoi_so_ct (
  loai_ct public.loai_ct not null,
  nam smallint not null,
  so_hien_tai integer not null default 0,
  primary key (loai_ct, nam)
);

create or replace function public.sinh_so_ct(p_loai public.loai_ct, p_nam smallint default null)
returns text
language plpgsql
as $$
declare
  v_so integer;
  v_nam smallint := coalesce(p_nam, extract(year from current_date)::smallint);
  v_tien_to text;
begin
  -- Một câu lệnh duy nhất: vừa tạo dòng đếm nếu chưa có, vừa tăng nếu đã có,
  -- vừa trả giá trị mới. Không có khoảng hở giữa đọc và ghi như cách SELECT rồi UPDATE.
  -- Hai transaction gọi đồng thời tự serialize trên khóa dòng (loai_ct, nam).
  insert into public.chuoi_so_ct (loai_ct, nam, so_hien_tai)
  values (p_loai, v_nam, 1)
  on conflict (loai_ct, nam)
  do update set so_hien_tai = public.chuoi_so_ct.so_hien_tai + 1
  returning so_hien_tai into v_so;

  v_tien_to := case p_loai
    when 'NHAP'       then 'PN'
    when 'XUAT'       then 'PX'
    when 'TRA_NCC'    then 'TN'
    when 'TRA_KHACH'  then 'TK'
    when 'CHUYEN_KHO' then 'CK'
    when 'KIEM_KE'    then 'KK'
    when 'DIEU_CHINH' then 'DC'
  end;

  return format('%s%s-%s', v_tien_to, to_char(v_nam % 100, 'FM00'), lpad(v_so::text, 6, '0'));
end;
$$;
```

Điểm bắt buộc:

- **Dùng `INSERT ... ON CONFLICT DO UPDATE ... RETURNING`**, KHÔNG dùng `SELECT MAX(so)+1`
  (race condition kinh điển) và KHÔNG dùng `UPDATE ... RETURNING` thuần (không xử lý được
  trường hợp năm/loại chưa có dòng đếm). Đây là điểm RESEARCH.md chỉnh lại so với D-18.
- **Không dùng Postgres sequence** — sequence không reset theo năm được (D-21).
- **Reset theo năm, không tách theo kho** (D-21): số `PN26-000001` dùng chung cả hai kho.
- `case` phải phủ đủ 7 giá trị enum, không có nhánh `else` — nếu thêm loại chứng từ mới,
  compiler sẽ không báo nhưng hàm trả NULL, nên viết đủ ngay từ đầu.

Kiểm nhanh sau khi reset:
```sql
select public.sinh_so_ct('NHAP', 2026::smallint);  -- PN26-000001
select public.sinh_so_ct('NHAP', 2026::smallint);  -- PN26-000002
select public.sinh_so_ct('XUAT', 2026::smallint);  -- PX26-000001
```
