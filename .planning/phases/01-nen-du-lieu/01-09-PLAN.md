```sql
-- Đặt ngữ cảnh phiên như thể người dùng đã đăng nhập với vai trò tương ứng.
-- Chỉ có tác dụng trong transaction hiện tại (tham số thứ 3 của set_config = true).
create or replace function public.dang_nhap_nhu(p_email text)
returns void
language plpgsql
as $$
declare
  v_user auth.users;
  v_nd public.nguoi_dung;
begin
  select * into v_user from auth.users where email = p_email;
  if v_user.id is null then
    raise exception 'Không có tài khoản mẫu %. Chạy npm run db:reset trước.', p_email;
  end if;
  select * into v_nd from public.nguoi_dung where id = v_user.id;

  perform set_config('request.jwt.claims',
    json_build_object(
      'sub', v_user.id::text,
      'role', 'authenticated',
      'vai_tro', v_nd.vai_tro::text,
      'kho_id', coalesce(v_nd.kho_id::text, '')
    )::text, true);
  perform set_config('role', 'authenticated', true);
end;
$$;

create or replace function public.dang_xuat()
returns void
language plpgsql
as $$
begin
  perform set_config('request.jwt.claims', '', true);
  perform set_config('role', 'postgres', true);
end;
$$;
```

Script dùng `@supabase/supabase-js` với `service_role` key của stack local (lấy từ
`npx supabase status`), gọi `supabase.auth.admin.createUser()` cho bốn tài khoản với

Yêu cầu:
- Đọc `SUPABASE_URL` và `SUPABASE_SERVICE_ROLE_KEY` từ biến môi trường, **không hardcode**.
- Ném lỗi rõ ràng nếu `SUPABASE_URL` không phải `localhost`/`127.0.0.1` — chặn chạy nhầm lên
  cloud. Đây là bảo vệ bắt buộc, không phải tùy chọn.
- Idempotent: tài khoản đã có thì bỏ qua, không ném lỗi.
- In ra bảng email → vai trò → kho sau khi chạy.
