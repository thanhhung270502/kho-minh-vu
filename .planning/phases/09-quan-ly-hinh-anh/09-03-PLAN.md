    ```ts
    export class ImageProcessingError extends Error {
      constructor(readonly title: string, readonly action: string) { super(`${title}. ${action}`); this.name = "ImageProcessingError"; }
    }
    export async function compressImage(file: File): Promise<{ full: Blob; thumb: Blob }>
    ```
    Luồng:
    1. `const problem = checkPickedFile(file); if (problem) throw new ImageProcessingError(problem.title, problem.action);`
    2. `createImageBitmap(file, { imageOrientation: "from-image" })` trong try/catch → lỗi → `ImageProcessingError("Không đọc được ảnh " + file.name, "Nếu là ảnh HEIC từ iPhone, chụp lại bằng nút Chụp ảnh hoặc xuất JPEG rồi chọn lại.")`.
    3. Hàm nội bộ `encode(bitmap, maxEdge, quality): Promise<Blob>`: `scaleToFit` → `document.createElement("canvas")`, gán width/height,
       `const ctx = canvas.getContext("2d"); if (!ctx) throw new ImageProcessingError("Trình duyệt không vẽ được ảnh", "Mở bằng Chrome hoặc Safari bản mới rồi thử lại.");`
       `ctx.drawImage(bitmap, 0, 0, w, h)`; `canvas.toBlob(cb, "image/webp", quality)` bọc Promise; `!blob || blob.type !== "image/webp"` →
       `ImageProcessingError("Trình duyệt này không nén được ảnh WebP", "Mở bằng Chrome hoặc Safari bản mới rồi thử lại.")`.
    4. `full = encode(bitmap, FULL_MAX_EDGE, FULL_QUALITY)`; nếu `full.size > MAX_FULL_BYTES` thử lại `FALLBACK_QUALITY`; vẫn vượt →
       `ImageProcessingError("Ảnh sau khi nén vẫn quá lớn", "Chụp lại gần hơn hoặc ảnh ít chi tiết hơn rồi thử lại.")`.
    5. `thumb = encode(bitmap, THUMB_MAX_EDGE, THUMB_QUALITY)`; vượt `MAX_THUMB_BYTES` → thử `FALLBACK_QUALITY`; vẫn vượt → lỗi như trên.
    6. `finally { bitmap.close(); }`. Trả `{ full, thumb }`.
    Import từ `./image-rules` (tương đối). Không `any`, không `!`, không `console.log`.
