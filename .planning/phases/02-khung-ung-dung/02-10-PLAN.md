export const VAI_TRO = ["quan_ly", "van_phong", "thu_kho", "chi_xem"] as const;
const tenDangNhap = z.string().trim().min(3, "Tên đăng nhập tối thiểu 3 ký tự").max(32)
  .transform(chuanHoaTenDangNhap).refine((v) => /^[a-z0-9._-]{3,32}$/.test(v), "Chỉ dùng chữ không dấu, số, dấu chấm, gạch");
