  email: z.union([z.literal(""), z.string().trim().email("Email không hợp lệ")]).optional().transform((v) => v || null),
  dia_chi: z.string().trim().optional().transform((v) => v || null),
  khu_vuc: z.string().trim().optional().transform((v) => v || null),
  ma_so_thue: z.string().trim().max(20).optional().transform((v) => v || null),
  ghi_chu: z.string().trim().optional().transform((v) => v || null),

});
