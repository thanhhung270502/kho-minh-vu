returns text language plpgsql volatile set search_path = '' as $$
declare v_tien_to text := case p_loai when 'NCC' then 'NCC' when 'KHACH' then 'KH' else 'DT' end;
        v_so int;
begin
  -- Mã ≥ 900000 là dải đặc biệt (NCC900001 = mã số thuế đổi ở Phase 1), không tính vào dãy.
  select coalesce(max(substring(ma from length(v_tien_to) + 1)::int), 0) + 1 into v_so
  from public.doi_tac

    and substring(ma from length(v_tien_to) + 1)::int < 900000;

end $$;
